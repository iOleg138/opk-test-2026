let answered = false;
let userAnswers = [];

const $ = id => document.getElementById(id);

function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function renderQuestion() {
  const q = currentQuestions[currentIndex];
  answered = false;
  $('questionNum').textContent = `Вопрос ${currentIndex + 1} из ${currentQuestions.length}`;
  $('questionText').textContent = q.q;
  $('explanation').className = 'explanation';
  $('explanation').textContent = '';
  $('nextBtn').classList.add('hidden');
  $('finishBtn').classList.add('hidden');
  $('progressText').textContent = `Вопрос ${currentIndex + 1} из ${currentQuestions.length}`;
  $('progressFill').style.width = `${(currentIndex / currentQuestions.length) * 100}%`;

  const optionsEl = $('options');
  optionsEl.innerHTML = '';
  const opts = shuffle(q.o);
  opts.forEach(opt => {
    const btn = document.createElement('button');
    btn.className = 'option';
    btn.innerHTML = `<span class="mark"></span><span>${opt}</span>`;
    btn.onclick = () => selectOption(btn, opt, q);
    optionsEl.appendChild(btn);
  });
}

function selectOption(btn, chosen, q) {
  if (answered) return;
  answered = true;

  const isCorrect = chosen === q.correct;
  document.querySelectorAll('.option').forEach(o => {
    o.classList.add('disabled');
    o.onclick = null;
    const text = o.querySelector('span:last-child').textContent;
    if (text === q.correct) {
      o.classList.add('correct');
      o.querySelector('.mark').textContent = '✓';
    } else if (text === chosen && !isCorrect) {
      o.classList.add('wrong');
      o.querySelector('.mark').textContent = '✗';
    }
  });

  if (isCorrect) score++; else wrong++;

  userAnswers.push({
    question: q.q,
    chosen: chosen,
    correct: q.correct,
    isCorrect: isCorrect,
    ex: q.ex || ''
  });

  if (mode === 'learn') {
    const exp = $('explanation');
    exp.classList.add('show');
    exp.classList.add(isCorrect ? 'correct' : 'wrong');
    let text = isCorrect ? '✅ Верно! ' : `❌ Неверно. Правильный ответ: ${q.correct}. `;
    if (q.ex) text += q.ex;
    exp.textContent = text;
  }

  $('scoreText').textContent = `Верно: ${score} · Ошибок: ${wrong}`;

  if (currentIndex < currentQuestions.length - 1) {
    $('nextBtn').classList.remove('hidden');
  } else {
    $('finishBtn').classList.remove('hidden');
  }
}

function nextQuestion() {
  if (currentIndex < currentQuestions.length - 1) {
    currentIndex++;
    renderQuestion();
  }
}

function finishQuiz() {
  $('quizContainer').classList.add('hidden');
  $('finishScreen').classList.remove('hidden');
  const total = currentQuestions.length;
  const percent = Math.round((score / total) * 100);
  $('finalScore').textContent = `${score} / ${total} (${percent}%)`;
  let comment = '';
  if (percent >= 90) comment = '🏆 Отлично! Ты готов к олимпиаде!';
  else if (percent >= 75) comment = '👍 Хорошо! Повтори сложные темы.';
  else if (percent >= 50) comment = '📚 Нормально, но стоит повторить материал.';
  else comment = '💪 Нужно ещё потренироваться. Попробуй режим обучения!';
  $('finalComment').textContent = comment;

  const reviewEl = $('review');
  if (mode === 'exam') {
    const mistakes = userAnswers.filter(a => !a.isCorrect);
    if (mistakes.length === 0) {
      reviewEl.innerHTML = '<p style="text-align:center;color:#558b2f;font-weight:600;">Все ответы верные! 🎉</p>';
    } else {
      reviewEl.innerHTML = '<h3 style="margin-top:20px;">Разбор ошибок:</h3>' +
        mistakes.map(m => `
          <div class="review-item wrong">
            <div class="q">${m.question}</div>
            <div class="a">Твой ответ: <span style="color:#c62828;">${m.chosen}</span></div>
            <div class="a correct-answer">Правильно: ${m.correct}</div>
            ${m.ex ? `<div class="a" style="margin-top:4px;font-size:13px;">${m.ex}</div>` : ''}
          </div>
        `).join('');
    }
  } else {
    reviewEl.innerHTML = '';
  }
}

function resetQuiz() {
  currentIndex = 0;
  score = 0;
  wrong = 0;
  userAnswers = [];
  $('quizContainer').classList.remove('hidden');
  $('finishScreen').classList.add('hidden');
  renderQuestion();
  $('scoreText').textContent = 'Верно: 0 · Ошибок: 0';
}

$('nextBtn').onclick = nextQuestion;
$('finishBtn').onclick = finishQuiz;
$('restartBtn').onclick = resetQuiz;
$('resetBtn').onclick = resetQuiz;

$('modeLearn').onclick = () => {
  mode = 'learn';
  $('modeLearn').className = 'btn-primary';
  $('modeExam').className = 'btn-secondary';
  resetQuiz();
};

$('modeExam').onclick = () => {
  mode = 'exam';
  $('modeExam').className = 'btn-primary';
  $('modeLearn').className = 'btn-secondary';
  resetQuiz();
};

$('shuffleBtn').onclick = () => {
  currentQuestions = shuffle(QUESTIONS);
  resetQuiz();
};

renderQuestion();
</script>
</body>
</html>
